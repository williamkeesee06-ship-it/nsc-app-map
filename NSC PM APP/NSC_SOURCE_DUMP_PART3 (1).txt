// ============================================================
// PRIVATE HELPERS
// ============================================================
function _extractRate(crewLabel) {
  const m = crewLabel.match(/\$(\d+)\/hr/);
  return m ? parseInt(m[1]) : null;
}

// ============================================================
// resolveSmartUnit — PRIMARY RESOLUTION FUNCTION
// Returns a normalized result object:
// {
//   color,         // hex string
//   label,         // optional canvas label
//   unit_code,     // primary billing code
//   unit,          // 'EA' | 'FT' | 'HRS'
//   qty,           // number
//   extraUnits,    // [{ unit_code, unit, qty }]
//   removeXMarks,  // boolean — line tools with REMOVE status
//   calloutText,   // string for transfer arrow callouts
//   calloutLines,  // string[] for T&E / DOWNTIME callout boxes
//   trenchDetail,  // object with trench detail fields (Change 6)
// }
// ============================================================
export function resolveSmartUnit(category, symbolKey, attributes = {}) {
  const catData = SMART_UNIT_DICTIONARY[category];
  if (!catData || !catData.logic || !catData.logic[symbolKey]) {
    return {
      unit_code: symbolKey,
      desc: symbolKey,
      unit: 'EA',
      qty: 1,
      color: '#000000',
      extraUnits: [],
    };
  }

  const logic = catData.logic[symbolKey];

  if (typeof logic.mapping !== 'function') {
    return {
      unit_code: symbolKey,
      desc: symbolKey,
      unit: 'EA',
      qty: 1,
      color: '#000000',
      extraUnits: [],
    };
  }

  const raw = logic.mapping(attributes);

  if (!raw) return null;

  // raw.units is an array of { code, type, qty, unit }
  const rawUnits = raw.units || [];

  if (rawUnits.length === 0) {
    // No billing (Existing, canvas-only, etc.)
    return {
      unit_code: null,
      desc: raw.label || symbolKey,
      unit: 'EA',
      qty: 0,
      color: raw.color || '#000000',
      label: raw.label || null,
      removeXMarks: raw.removeXMarks || false,
      calloutText:  raw.calloutText  || null,
      calloutLines: raw.calloutLines || null,
      dashArray:    raw.dashArray    || null,
      trenchDetail: raw.trenchDetail || null,
      extraUnits: [],
    };
  }

  const primary = rawUnits[0];
  const extras  = rawUnits.slice(1).map(u => ({
    unit_code: u.unit_code || u.code,
    desc:      u.desc      || u.unit_code || u.code,
    unit:      u.unit      || 'FT',
    qty:       u.qty       != null ? u.qty : 1,
  }));

  return {
    unit_code:    primary.unit_code || primary.code,
    desc:         primary.desc      || primary.unit_code || primary.code,
    unit:         primary.unit      || 'FT',
    qty:          primary.qty       != null ? primary.qty : 1,
    color:        raw.color    || '#000000',
    label:        raw.label    || null,
    removeXMarks: raw.removeXMarks || false,
    calloutText:  raw.calloutText  || null,
    calloutLines: raw.calloutLines || null,
    dashArray:    raw.dashArray    || null,
    billDashCode:  raw.billDashCode  || null,
    billDashHours: raw.billDashHours || null,
    trenchDetail:  raw.trenchDetail  || null,
    extraUnits:   extras,
  };
}

  if (['RMV_BURIED_FAC'].includes(symbolKey)) category = 'underground';
  if (['RMV_ANCHOR'].includes(symbolKey)) category = 'aerial';
  if (['TNE','DOWNTIME','SPLICER_FIBER','SPLICER_COPPER','EMERGENCY_TRAVEL','VAC_TRUCK','CORE_DRILL'].includes(symbolKey)) category = 'misc';
  return resolveSmartUnit(category, symbolKey, attributes);
}

import * as THREE from 'three';
import { resolveUnit, resolveSmartUnit, SMART_UNIT_DICTIONARY, ASBUILT_THEME } from './units.js';

export class Sketcher {
    constructor(app) {
        this.app = app;
        this.raycaster = new THREE.Raycaster();
        this.mouse = new THREE.Vector2();

        // Interaction State
        this.isDrawing = false;
        this.currentLine = null; // Temporary line mesh while drawing
        this.startPoint = null; // Vector3

        // Snapping
        this.snapRadius = 1.0; // Distance to snap to existing vertices
        this.snapRadius = 1.0; // Distance to snap to existing vertices
        this.gridSize = 10; // Grid snap

        // Selection / Transformation State
        this.interactionMode = 'IDLE'; // 'IDLE', 'MOVE', 'RESIZE'
        this.activeHandle = null; // 'tl', 'tr', 'bl', 'br'
        this.dragStart = new THREE.Vector3(); // Mouse World Pos at start
        this.objectStartPos = new THREE.Vector3(); // Object Pos at start
        this.objectStartScale = new THREE.Vector3(); // Object Scale at start
        this.selectionHelper = null; // Group { box, handles... }

        this.initSelectionHelper();

        // Helper marker
        this.cursorMarker = this.createCursorMarker();
        this.app.scene.add(this.cursorMarker);

        // Events
        // Events
        this.app.renderer.domElement.addEventListener('mousemove', (e) => this.onMouseMove(e));
        this.app.renderer.domElement.addEventListener('mousedown', (e) => this.onMouseDown(e));
        this.app.renderer.domElement.addEventListener('mouseup', (e) => this.onMouseUp(e));
        this.app.renderer.domElement.addEventListener('mouseup', (e) => this.onMouseUp(e));

        // Global Keydown for Tools
        window.addEventListener('keydown', (e) => this.onKeyDown(e));
    }

    initSelectionHelper() {
        this.selectionHelper = new THREE.Group();
        this.selectionHelper.name = 'SelectionHelper';
        this.selectionHelper.visible = false;

        // 1. Bounding Box (Blue Line)
        const boxGeom = new THREE.BufferGeometry();
        const boxMat = new THREE.LineBasicMaterial({ color: 0x00a5e6, depthTest: false, transparent: true });
        this.selectionBox = new THREE.LineSegments(boxGeom, boxMat);
        this.selectionBox.renderOrder = 999;
        this.selectionHelper.add(this.selectionBox);

        // 2. Resize Handles
        this.handles = {};
        const handleKeys = ['tl', 'tr', 'bl', 'br'];
        const handleGeom = new THREE.PlaneGeometry(2, 2);
        const handleMat = new THREE.MeshBasicMaterial({ color: 0x00a5e6, depthTest: false, transparent: true });

        handleKeys.forEach(key => {
            const handle = new THREE.Mesh(handleGeom, handleMat);
            handle.rotation.x = -Math.PI / 2; // Flat
            handle.renderOrder = 1000;
            handle.name = `handle_${key}`;
            this.handles[key] = handle;
            this.selectionHelper.add(handle);
        });

        this.app.scene.add(this.selectionHelper);
    }

    updateSelectionHelper() {
        const obj = this.app.selectedObject?.mesh;
        if (!obj) {
            this.selectionHelper.visible = false;
            return;
        }

        this.selectionHelper.visible = true;

        // Calculate Bounds
        const box = new THREE.Box3().setFromObject(obj);
        const min = box.min;
        const max = box.max;
        const y = 0.6;

        // Handles Positions (Top-Down View: X/Z plane)
        const tl = new THREE.Vector3(min.x, y, min.z);
        this.handles.tl.position.set(min.x, y, min.z);
        this.handles.tr.position.set(max.x, y, min.z);
        this.handles.bl.position.set(min.x, y, max.z);
        this.handles.br.position.set(max.x, y, max.z);

        // Update Box Geometry points
        const points = [
            this.handles.tl.position, this.handles.tr.position,
            this.handles.tr.position, this.handles.br.position,
            this.handles.br.position, this.handles.bl.position,
            this.handles.bl.position, this.handles.tl.position
        ];
        this.selectionBox.geometry.setFromPoints(points);
    }

    onKeyDown(event) {
        if (event.key === 'Escape') {
            this.interactionMode = 'IDLE';
            this.app.selectObject(null);
            this.updateSelectionHelper();
        }
    }

    syncObjectData() {
        if (!this.app.selectedObject) return;
        const obj = this.app.selectedObject;
        const mesh = obj.mesh;

        obj.position = mesh.position.clone();

        // Update Length for Lines if scaled
        if (obj.type.includes('LINE') || obj.type.includes('CABLE')) {
            obj.length = mesh.scale.x;
        }

        this.app.updateTotals();
    }

    createCursorMarker() {
        const geometry = new THREE.SphereGeometry(0.5, 16, 16);
        const material = new THREE.MeshBasicMaterial({ color: 0xffff00, opacity: 0.5, transparent: true });
        return new THREE.Mesh(geometry, material);
    }

    getIntersects(event) {
        const rect = this.app.renderer.domElement.getBoundingClientRect();
        this.mouse.x = ((event.clientX - rect.left) / rect.width) * 2 - 1;
        this.mouse.y = -((event.clientY - rect.top) / rect.height) * 2 + 1;

        this.raycaster.setFromCamera(this.mouse, this.app.camera);

        // Intersect with ground
        return this.raycaster.intersectObject(this.app.ground);
    }

    getSnappedPosition(rawPos) {
        // 1. Vertex Snapping (Snap to existing objects)
        let closestDist = Infinity;
        let snapPos = null;

        for (const obj of this.app.drawnObjects) {
            // Check start and end points of lines, or position of points
            const pointsToCheck = [];
            if (obj.type === 'LINE') {
                pointsToCheck.push(obj.start, obj.end);
            } else if (obj.type === 'POINT') {
                pointsToCheck.push(obj.position);
            }

            for (const p of pointsToCheck) {
                const dist = rawPos.distanceTo(p);
                if (dist < this.snapRadius && dist < closestDist) {
                    closestDist = dist;
                    snapPos = p.clone();
                }
            }
        }

        if (snapPos) return snapPos;

        // 2. Grid Snapping (if no vertex snap)
        // Only snap X and Z on the grid, keep Y (height) free or snapped to levels? 
        // For now, simple grid snap on X/Z
        const snappedX = Math.round(rawPos.x / this.gridSize) * this.gridSize;
        const snappedZ = Math.round(rawPos.z / this.gridSize) * this.gridSize;
        return new THREE.Vector3(snappedX, rawPos.y, snappedZ);
    }

    getDrawnObjectIntersects(event) {
        const rect = this.app.renderer.domElement.getBoundingClientRect();
        this.mouse.x = ((event.clientX - rect.left) / rect.width) * 2 - 1;
        this.mouse.y = -((event.clientY - rect.top) / rect.height) * 2 + 1;
        this.raycaster.setFromCamera(this.mouse, this.app.camera);
        this.raycaster.params.Line.threshold = 3;

        const meshes = this.app.drawnObjects.map(o => o.mesh).filter(m => m);
        return this.raycaster.intersectObjects(meshes, true);
    }

    getHandleIntersects() {
        if (!this.selectionHelper.visible) return [];
        const handleMeshes = Object.values(this.handles);
        return this.raycaster.intersectObjects(handleMeshes);
    }



    // 1. Mouse Down: Select or Start Drawing
    onMouseDown(event) {
        if (event.target !== this.app.renderer.domElement) return;

        // 1. Check Handles (Resize)
        const handleIntersects = this.getHandleIntersects();
        if (handleIntersects.length > 0) {
            const handle = handleIntersects[0].object;
            const handleName = handle.name.replace('handle_', '');
            this.interactionMode = 'RESIZE';
            this.activeHandle = handleName;

            // Interaction Start State
            const intersects = this.getIntersects(event);
            this.dragStart.copy(intersects[0].point);
            const obj = this.app.selectedObject.mesh;
            this.objectStartPos.copy(obj.position);
            this.objectStartScale.copy(obj.scale);

            this.app.controls.enabled = false;
            return;
        }

        // 2. Check Objects (Selection / Move)
        const objectIntersects = this.getDrawnObjectIntersects(event);
        if (objectIntersects.length > 0) {
            // Pass through TransformControls first?
            // If we clicked the Gizmo, ThreeJS Raycaster handles it via controls logic usually, 
            // but we need to ensure we don't deselect if we clicked the gizmo.
            // Actually, `getDrawnObjectIntersects` filters for drawings.

            const clickedMesh = objectIntersects[0].object;
            const parentItem = this.app.drawnObjects.find(obj =>
                obj.mesh === clickedMesh ||
                (obj.mesh.children && obj.mesh.children.includes(clickedMesh))
            );
            if (parentItem) {
                // If already selected, maybe dragging?
                if (this.app.selectedObject === parentItem) {
                    this.interactionMode = 'MOVE';

                    const intersects = this.getIntersects(event);
                    if (intersects.length > 0) this.dragStart.copy(intersects[0].point);
                    this.objectStartPos.copy(parentItem.mesh.position);
                    this.app.controls.enabled = false;
                } else {
                    // New Selection
                    this.app.selectObject(parentItem);
                    this.updateSelectionHelper();

                    // Enable Move immediately
                    this.interactionMode = 'MOVE';
                    const intersects = this.getIntersects(event);
                    if (intersects.length > 0) this.dragStart.copy(intersects[0].point);
                    this.objectStartPos.copy(parentItem.mesh.position);
                    this.app.controls.enabled = false;
                }
                return;
            }
        }

        // If we are hovering over the gizmo, do not start drawing!


        // If we didn't hit a drawn object, and we are not on gizmo:
        if (objectIntersects.length === 0) {
            // If we clicked nothing, deselect
            this.app.selectObject(null);
            this.updateSelectionHelper();
            this.interactionMode = 'IDLE';
        }

        const position = this.cursorMarker.position.clone();

        if (this.app.activeTool === 'POINT') {
            const category = this.app.activeCategory;
            const key = this.app.selectedUnitKey;
            const catData = SMART_UNIT_DICTIONARY[category];
            const symbol = catData?.symbols.find(s => s.key === key);
            const shape = symbol?.shape || 'SQUARE';

            // 1. Render Marker Immediately (Optimistic UI)
            const previewMarker = this.createSymbolSprite(shape, '#0284c7', '...');
            previewMarker.position.copy(position);
            previewMarker.position.y = 0.5;
            previewMarker.material.opacity = 0.9;
            this.app.scene.add(previewMarker);

            // 2. Trigger Modal
            this.app.showSmartModal(category, key, (attributes) => {
                this.app.scene.remove(previewMarker); // Clear preview 
                if (attributes) {
                    // 3. Save officially
                    this.placePoint(position, attributes);
                }
            });
        } else if (['LINE', 'RECT', 'CIRCLE', 'ARROW', 'PENCIL', 'STRAND', 'TRENCH', 'BORE', 'ANCHOR', 'DIMENSION', 'CALLOUT'].includes(this.app.activeTool)) {
            this.startDrawing(position);
        }
        else if (this.app.activeTool === 'TEXT') {
            const text = prompt("Enter label text:", "Label");
            if (text) this.placeText(position, text);
        }
    }

    placeImage(texture) {
        const geometry = new THREE.PlaneGeometry(20, 20); // Default size 20ft
        const material = new THREE.MeshBasicMaterial({ map: texture, transparent: true });
        const mesh = new THREE.Mesh(geometry, material);
        mesh.rotation.x = -Math.PI / 2;
        mesh.position.set(0, 0.1, 0); // Flat on ground
        this.app.scene.add(mesh);

        this.app.drawnObjects.push({
            type: 'IMAGE',
            category: 'basic',
            baseKey: 'BASIC_IMAGE',
            mesh: mesh,
            color: '#ffffff'
        });
        this.app.updateTotals();
    }

    // 2. Mouse Move: Update Cursor and Shape Preview
    onMouseMove(event) {
        // Handle Interactions
        if (this.interactionMode === 'MOVE') {
            const intersects = this.getIntersects(event);
            if (intersects.length > 0) {
                const currentPos = intersects[0].point;
                const delta = new THREE.Vector3().subVectors(currentPos, this.dragStart);

                const obj = this.app.selectedObject.mesh;
                obj.position.copy(this.objectStartPos).add(delta);
                this.updateSelectionHelper();
            }
            return;
        }
        else if (this.interactionMode === 'RESIZE') {
            const intersects = this.getIntersects(event);
            if (intersects.length > 0) {
                const currentPos = intersects[0].point;

                const deltaX = currentPos.x - this.dragStart.x;
                const deltaZ = currentPos.z - this.dragStart.z;

                const obj = this.app.selectedObject.mesh;

                const factor = 1.0; // 1:1 movement
                let sx = this.objectStartScale.x;
                let sy = this.objectStartScale.y;

                // Note: This logic assumes Uniform Center Scaling or similar.
                // For true PDF dragging, we want edge dragging.
                // But for this simple mesh scale, we will just scale X/Y relative to start.

                if (this.activeHandle === 'br') {
                    sx += deltaX * factor;
                    sy += deltaZ * factor;
                } else if (this.activeHandle === 'bl') {
                    sx -= deltaX * factor;
                    sy += deltaZ * factor;
                } else if (this.activeHandle === 'tr') {
                    sx += deltaX * factor;
                    sy -= deltaZ * factor;
                } else if (this.activeHandle === 'tl') {
                    sx -= deltaX * factor;
                    sy -= deltaZ * factor;
                }

                sx = Math.max(0.1, sx);
                sy = Math.max(0.1, sy);

                obj.scale.set(sx, sy, 1);
                this.updateSelectionHelper();
            }
            return;
        }

        // Hover Logic
        const handleIntersects = this.getHandleIntersects();
        if (handleIntersects.length > 0) {
            document.body.style.cursor = 'nwse-resize';
        } else {
            const objectIntersects = this.getDrawnObjectIntersects(event);
            if (objectIntersects.length > 0) {
                if (this.app.selectedObject && objectIntersects[0].object === this.app.selectedObject.mesh) {
                    document.body.style.cursor = 'move';
                } else {
                    document.body.style.cursor = 'pointer';
                }
            } else {
                document.body.style.cursor = 'default';
            }
        }

        const intersects = this.getIntersects(event);
        if (intersects.length > 0) {
            const rawPos = intersects[0].point;
            rawPos.y = 0;
            const snapped = this.getSnappedPosition(rawPos);
            this.cursorMarker.position.copy(snapped);

            // Update Preview
            if (this.isDrawing && this.currentShape) {
                this.updateShapePreview(snapped);
            }
        }
    }

    // 3. Mouse Up: Finish Shape
    onMouseUp(event) {
        if (this.interactionMode === 'MOVE' || this.interactionMode === 'RESIZE') {
            this.interactionMode = 'IDLE';
            this.app.controls.enabled = true;
            this.syncObjectData();
            return;
        }

        if (this.isDrawing) {
            const position = this.cursorMarker.position.clone();
            this.finishDrawing(position);
        }
    }

    placePoint(position, attributes = {}) {
        const category = this.app.activeCategory;
        const key = this.app.selectedUnitKey;
        const unitData = resolveSmartUnit(category, key, attributes);

        let color = ASBUILT_THEME.EXISTING; // Default

        // Use Logic Color if available
        if (unitData.color) {
            color = unitData.color;
        } else {
            if (attributes.condition === 'Remove') color = ASBUILT_THEME.REMOVE;
            else if (attributes.condition === 'New') color = ASBUILT_THEME.NEW;
            else if (this.app.userColor) color = this.app.userColor;
        }

        const catData = SMART_UNIT_DICTIONARY[category];
        const symbol = catData?.symbols.find(s => s.key === key);
        const shape = symbol?.shape || 'SQUARE';

        // Use Logic Label if available (e.g. "A123 35'")
        let labelText = unitData.label || unitData.unit_code;
        if (attributes.tag && (shape === 'RECT_LABEL' || shape === 'HH_BOX' || shape === 'PED' || shape === 'SQUARE_DOT')) {
            labelText = attributes.tag;
        }
        // Force Tag Only for Pole (User Request: "ONLY THE ATAG AND NO OTHER INFO")
        if (shape === 'POLE_X') {
            if (attributes.ATAG) {
                labelText = attributes.ATAG;
            } else if (labelText && typeof labelText === 'string') {
                // Fallback: split "A999 45'" -> "A999"
                labelText = labelText.split(' ')[0];
            }
        }

        const sprite = this.createSymbolSprite(shape, color, labelText);
        sprite.position.copy(position);
        sprite.position.y = 0.5;

        // SCALE ADJUSTMENT FOR VISIBILITY
        if (shape === 'POLE_X') {
            sprite.scale.set(30, 30, 1); // VERY LARGE Pole - Square Scale
        } else {
            sprite.scale.set(4, 4, 1); // Standard scale for others
        }

        this.app.scene.add(sprite);

        // Render Tag if present (e.g. Existing Pole Tag)
        let tagSprite = null;
        if (attributes.tag) {
            tagSprite = this.createTextSprite(attributes.tag, color); // Pass Symbol Color
            tagSprite.position.copy(position);
            tagSprite.position.z -= 10; // Offset "up" significantly (North)
            tagSprite.position.y = 5;   // Higher z-index visually? No, y is up. Lift it up so it doesn't clip.
            this.app.scene.add(tagSprite);
        }

        this.app.drawnObjects.push({
            type: 'POINT',
            category: category,
            baseKey: key,
            attributes: attributes,
            unit_code: unitData.unit_code,
            position: position.clone(),
            mesh: sprite,
            tagMesh: tagSprite // Store for undo/cleanup
        });

        this.app.updateTotals();
    }

    startDrawing(position) {
        this.isDrawing = true;
        this.startPoint = position;

        // --- GLOBAL DEFAULTS ENFORCEMENT ---
        // If no explicit override, default to Black and Weight 2
        let color = this.app.userColor || ASBUILT_THEME.EXISTING;

        // Status Override (Preview) removed - strictly modal based now
        // if (this.app.globalStatus === 'New') color = ASBUILT_THEME.NEW; ...

        let weight = this.app.userWeight || 2;

        if (this.app.selectedUnitKey === 'TRENCH') {
            color = '#FF0000';
            weight = 3;
        }

        // STRAND Force Thickness 3 (User Request)
        if (this.app.selectedUnitKey === 'STRAND_10M' || this.app.selectedUnitKey === 'STRAND') {
            weight = 3;
        }



        // Removal Tool Pre-selection Color Override
        if (this.app.selectedUnitKey && this.app.selectedUnitKey.startsWith('RMV_')) {
            color = ASBUILT_THEME.REMOVE;
        }

        const key = this.app.selectedUnitKey;
        const isStrand = (key === 'STRAND' || key === 'STRAND_10M');

        if (['LINE', 'PENCIL', 'STRAND', 'TRENCH', 'BORE', 'DIMENSION', 'CALLOUT'].includes(this.app.activeTool)) {
            // Special handling for Cables (including Removal) -> Group { Line, Markers }
            // ALSO handling                const key = this.app.selectedUnitKey;
            const isCable = key === 'COPPER_CABLE' || key === 'FIBER_CABLE' || (key && key.startsWith('RMV_'));
            // Treat TRENCH and BORE as "Strand-like" (Mesh Lines) for thickness support
            const isStrand = (key === 'STRAND' || key === 'STRAND_10M');
            const isTrench = key === 'TRENCH';
            const isBore = key === 'BORE';
            const isMeshLine = isCable || isStrand || isTrench || isBore;

            if (isMeshLine) {
                this.currentShape = new THREE.Group();
                this.currentShape.name = isStrand ? 'STRAND_GROUP' : 'CABLE_GROUP'; // Reuse Group Name or generic? 
                // Let's stick to CABLE_GROUP for general mesh-lines unless it needs specific handling

                if (key.startsWith('RMV_')) {
                    color = ASBUILT_THEME.REMOVE;
                } else if (isStrand) {
                    color = this.app.userColor || '#000000';
                    if (this.app.userColor === '#0284c7') color = '#000000';
                } else if (isTrench || isBore) {
                    color = '#8B4513'; // Construction Brown
                } else {
                    color = ASBUILT_THEME.EXISTING;
                }

                // 1. The Line (Mesh for thickness)
                const geometry = new THREE.PlaneGeometry(1, 1);
                geometry.translate(0.5, 0, 0); // Anchor at start

                let material;
                if (isStrand || isBore) {
                    const dashTexture = this.createDashedTexture();
                    material = new THREE.MeshBasicMaterial({
                        color: color,
                        side: THREE.DoubleSide,
                        map: dashTexture,
                        transparent: true,
                        alphaTest: 0.1
                    });
                } else {
                    material = new THREE.MeshBasicMaterial({ color: color, side: THREE.DoubleSide });
                }

                const line = new THREE.Mesh(geometry, material);
                line.rotation.x = -Math.PI / 2; // Flat on ground
                line.position.y = 0.5;
                line.name = 'line_mesh';
                this.currentShape.add(line);

                // 2. The Markers Group (Only for Cables)
                if (isCable) {
                    const markers = new THREE.Group();
                    markers.name = 'markers_group';
                    this.currentShape.add(markers);
                }
            }
            else {
                // Standard Line logic
                const geometry = new THREE.BufferGeometry();
                const vertices = new Float32Array([position.x, 0.5, position.z, position.x, 0.5, position.z]);
                geometry.setAttribute('position', new THREE.BufferAttribute(vertices, 3));

                let material;
                if (this.app.activeTool === 'STRAND') {
                    // Default to Black if user hasn't picked a custom color (assuming Blue #0284c7 is default)
                    // Otherwise use the user's selected color.
                    const isDefaultColor = !this.app.userColor || this.app.userColor === '#0284c7';
                    const strandColor = isDefaultColor ? 0x000000 : this.app.userColor;

                    material = new THREE.LineDashedMaterial({
                        color: strandColor,
                        linewidth: weight,
                        scale: 1,
                        dashSize: 4, // Increased slightly for visibility
                        gapSize: 3
                    });
                } else {
                    material = new THREE.LineBasicMaterial({ color: color, linewidth: weight });
                }

                this.currentShape = new THREE.Line(geometry, material);
                if (this.app.activeTool === 'STRAND') this.currentShape.computeLineDistances();
            }
        }
        else if (this.app.activeTool === 'ARROW') {
            // Arrow is a Group: Line + Cone
            this.currentShape = new THREE.Group();

            const lineGeom = new THREE.BufferGeometry().setAttribute('position', new THREE.BufferAttribute(new Float32Array([0, 0, 0, 0, 0, 0]), 3));
            const lineMat = new THREE.LineBasicMaterial({ color: color, linewidth: weight });
            const line = new THREE.Line(lineGeom, lineMat);
            line.name = 'arrow_line';

            const headGeom = new THREE.ConeGeometry(1, 4, 8);
            const headMat = new THREE.MeshBasicMaterial({ color: color });
            const head = new THREE.Mesh(headGeom, headMat);
            head.name = 'arrow_head';
            head.visible = false; // Hide until we have direction

            this.currentShape.add(line, head);
        }
        else if (this.app.activeTool === 'ANCHOR') {
            // Default Black if not user override
            const isDefaultColor = !this.app.userColor || this.app.userColor === '#0284c7';
            const anchorColor = isDefaultColor ? '#000000' : this.app.userColor;

            // Dynamic look up of shape/icon from dictionary
            const category = this.app.activeCategory;
            const key = this.app.selectedUnitKey;
            const catData = SMART_UNIT_DICTIONARY[category];
            const symbol = catData?.symbols.find(s => s.key === key);
            const shape = symbol?.shape || 'ANCHOR_ARROW';
            const icon = symbol?.icon || '--<'; // Fallback

            this.currentShape = this.createSymbolSprite(shape, anchorColor, icon);
            this.currentShape.position.copy(position);
            this.currentShape.position.y = 0.5;

            // Re-creating as a Mesh Plane for rotation capability
            const texture = this.currentShape.material.map;
            const material = new THREE.MeshBasicMaterial({ map: texture, transparent: true, color: anchorColor });
            // Aspect ratio? The sprite was 10x10. Let's make a plane 10x10.
            const geometry = new THREE.PlaneGeometry(10, 10);
            const mesh = new THREE.Mesh(geometry, material);
            mesh.rotation.x = -Math.PI / 2; // Flat on ground
            mesh.position.copy(position);
            mesh.position.y = 0.5;

            // Let's manually create the anchor mesh here.
            this.currentShape = mesh;
        }
        else if (this.app.activeTool === 'RECT') {
            this.currentShape = new THREE.Group();
            this.currentShape.name = 'RECT_GROUP';

            // Visual Line
            const points = [
                new THREE.Vector3(position.x, 0.5, position.z),
                new THREE.Vector3(position.x, 0.5, position.z),
                new THREE.Vector3(position.x, 0.5, position.z),
                new THREE.Vector3(position.x, 0.5, position.z)
            ];
            const geometry = new THREE.BufferGeometry().setFromPoints(points);
            const material = new THREE.LineBasicMaterial({ color: color, linewidth: weight });
            const line = new THREE.LineLoop(geometry, material);
            line.name = 'rect_visual';
            this.currentShape.add(line);

            // Hit Plane (for easy selection)
            const planeGeom = new THREE.PlaneGeometry(1, 1);
            const planeMat = new THREE.MeshBasicMaterial({ color: 0xffffff, transparent: true, opacity: 0.01, side: THREE.DoubleSide, depthTest: false });
            const plane = new THREE.Mesh(planeGeom, planeMat);
            plane.rotation.x = -Math.PI / 2;
            plane.name = 'rect_hit_plane';
            plane.visible = false;
            this.currentShape.add(plane);
        }
        else if (this.app.activeTool === 'CIRCLE') {
            const segmentCount = 64;
            const points = new Array(segmentCount).fill(new THREE.Vector3(position.x, 0.5, position.z));
            const geometry = new THREE.BufferGeometry().setFromPoints(points);
            const material = new THREE.LineBasicMaterial({ color: color, linewidth: weight });
            this.currentShape = new THREE.LineLoop(geometry, material);
        }
        else if (this.app.activeTool === 'PENCIL') {
            const points = [new THREE.Vector3(position.x, 0.5, position.z)];
            const geometry = new THREE.BufferGeometry().setFromPoints(points);
            const material = new THREE.LineBasicMaterial({ color: color, linewidth: weight });
            this.currentShape = new THREE.Line(geometry, material);
            this.currentShape.userData.points = points;
        }

        if (this.currentShape) this.app.scene.add(this.currentShape);
    }

    updateShapePreview(currentPos) {
        const start = this.startPoint;
        if (!this.currentShape) return;
        const key = this.app.selectedUnitKey;

        if (['LINE', 'STRAND', 'TRENCH', 'BORE'].includes(this.app.activeTool)) {
            const isCable = key === 'COPPER_CABLE' || key === 'FIBER_CABLE' || (key && key.startsWith('RMV_'));
            const isStrand = (key === 'STRAND' || key === 'STRAND_10M');
            const isTrench = key === 'TRENCH';
            const isBore = key === 'BORE';
            const isMeshLine = isCable || isStrand || isTrench || isBore;

            if (isMeshLine) {

                // Handle Group: line_mesh, markers_group (Strand has no markers group but logic safe check)
                const line = this.currentShape.getObjectByName('line_mesh');
                const markers = this.currentShape.getObjectByName('markers_group'); // Undefined for Strand

                // Update Line Mesh (Scale and Rotate)
                const dx = currentPos.x - start.x;
                const dz = currentPos.z - start.z;
                const dist = Math.sqrt(dx * dx + dz * dz) || 0.001;

                // Scale: X = Length, Y = Thickness
                let thickness = 0.5 + ((this.app.userWeight || 2) * 0.2);

                if (key.includes('STRAND')) thickness = 1.5;
                if (key === 'TRENCH') thickness = 2.0; // Thickness 4 -> ~2.0 units
                if (key === 'BORE') thickness = 1.5;   // Thickness 3 -> ~1.5 units

                line.scale.set(dist, thickness, 1);

                // Update Texture Repeat for Strand/Bore
                if ((isStrand || isBore) && line.material.map) {
                    // Repeat texture every 4 units
                    const repeat = dist / 4;
                    line.material.map.repeat.set(repeat, 1);
                }

                // Position: Midpoint of the line
                // The Mesh is centered at origin (0,0) of local space?
                // Geometry was: PlaneGeometry(1, 1).translate(0.5, 0, 0); 
                // So origin is at Left edge.

                line.position.copy(start);
                line.position.y = 0.5;

                // Rotation:
                // Default: +X direction.
                // We want +X to point to currentPos.
                // Angle atan2(dz, dx) is angle from +X (CCW).
                // Rotation Y = -angle.
                // Mesh is flat on ground (Rotation X = -90).
                // Initial: Normal +Y (Up). Local X = World X. Local Y = World -Z.
                // We rotate around World Y.
                // Since object is rotated X=-90, Local Z axis is World Y axis?
                // X cross Y -> Z.
                // X (Right), Y (Up), Z (Forward).
                // Rot X -90:
                // New Y axis points along -Z (Away).
                // New Z axis points along +Y (Up).
                // So rotating around Local Z rotates around World Y.
                // Rotation is CCW.
                // Angle `atan2(dz, dx)` is standard math angle (CCW from +X).
                // So `line.rotation.z = angle` should work if Z is Up.

                // Let's reset rotation completely to ensure cleanliness
                line.rotation.set(-Math.PI / 2, 0, 0);
                line.rotateZ(Math.atan2(dz, dx));
                // Note: reduce atan2(dz, dx) to flip if needed? 
                // Check coordinate system: 
                // +Z in ThreeJS is towards viewer (Screen).
                // +X Right.
                // atan2(dz, dx): (0, 1) -> +Z -> PI/2.
                // Rotate Z by PI/2 (90 deg CCW).
                // Original X axis -> Points along Local Y? 
                // Local Y was World -Z.
                // So X -> -Z.
                // But we want X -> +Z.
                // So we need -Math.atan2(dz, dx)? 

                // Try: -angle
                // (0, 1) -> angle PI/2. -PI/2 = -90.
                // X -> -90 deg -> Points +Z? 
                // Standard Right Hand Rule around Z (Up): X -> Y.
                // Local X -> Local Y (-World Z). 
                // So +90 means X points Away (-Z).
                // We want X points Towards (+Z).
                // So -90.
                // Thus: -angle works.
                line.rotation.set(-Math.PI / 2, 0, -Math.atan2(dz, dx));

                // Update Markers (Cables Only)
                if (markers) {
                    markers.clear();
                    const spacing = 40; // Every 40 units
                    const count = Math.floor(dist / spacing);

                    if (count > 0) {
                        const dir = new THREE.Vector3().subVectors(currentPos, start).normalize();
                        if (key.startsWith('RMV_')) previewColor = ASBUILT_THEME.REMOVE;

                        const symbolShape = (key.includes('COPPER')) ? 'COPPER_C' : 'FIBER_F';

                        // Lookup Symbol Data for Custom Text (e.g. X-AC-X)
                        let customText = null;
                        const catData = SMART_UNIT_DICTIONARY[this.app.activeCategory];
                        const symbolData = catData?.symbols.find(s => s.key === key);

                        if (symbolData && symbolData.icon) {
                            customText = symbolData.icon;
                        }

                        for (let i = 1; i <= count; i++) {
                            const sprite = this.createSymbolSprite(symbolShape, previewColor, customText);
                            const pos = start.clone().add(dir.clone().multiplyScalar(i * spacing));
                            sprite.position.copy(pos);
                            sprite.position.y = 0.75; // Left up to verify visibility
                            sprite.scale.set(16, 16, 1); // Double size
                            // Rotate sprite to match line angle?
                            // Sprites always face camera usually. Or flat.
                            // Existing createSymbolSprite makes a sprite (billboard).
                            // If we want flat text on ground, we need a Mesh, not Sprite.
                            // createSymbolSprite returns a Sprite?
                            // Let's check definition. It returns THREE.Sprite or Mesh?
                            // It returns a Sprite (Material uses canvas).
                            // Sprites always face camera.
                            // If user wants them strictly on the line, Sprite is okay but "flat" text is better.
                            // But let's stick to Sprite for compatibility with existing code unless requested.
                            markers.add(sprite);
                        }
                    }
                }
                else {
                    const positions = this.currentShape.geometry.attributes.position.array;
                    positions[3] = currentPos.x;
                    positions[4] = 0.5;
                    positions[5] = currentPos.z;
                    this.currentShape.geometry.attributes.position.needsUpdate = true;
                    if (this.app.activeTool === 'STRAND') {
                        this.currentShape.computeLineDistances();
                    }
                }
            }
        }
        else if (this.app.activeTool === 'DIMENSION') {
            // Basic Line with End ticks
            // Geometry needs to be dynamic. 
            // Logic: Draw line from Start to Current.
            // Visuals: Center text is handled in finishDrawing usually, but we could preview?
            // Just draw the line and cross-bars for now.

            const positions = this.currentShape.geometry.attributes.position.array;
            positions[0] = start.x; positions[1] = 0.5; positions[2] = start.z;
            positions[3] = currentPos.x; positions[4] = 0.5; positions[5] = currentPos.z;
            this.currentShape.geometry.attributes.position.needsUpdate = true;
        }
        else if (this.app.activeTool === 'CALLOUT') {
            // Line from Start (Target) to Current (Text Box location)
            const positions = this.currentShape.geometry.attributes.position.array;
            positions[0] = start.x; positions[1] = 0.5; positions[2] = start.z;
            positions[3] = currentPos.x; positions[4] = 0.5; positions[5] = currentPos.z;
            this.currentShape.geometry.attributes.position.needsUpdate = true;
        }
        else if (this.app.activeTool === 'ARROW') {
            const line = this.currentShape.getObjectByName('arrow_line');
            const head = this.currentShape.getObjectByName('arrow_head');

            // Update Line
            const positions = line.geometry.attributes.position.array;
            positions[0] = start.x; positions[1] = 0.5; positions[2] = start.z;
            positions[3] = currentPos.x; positions[4] = 0.5; positions[5] = currentPos.z;
            line.geometry.attributes.position.needsUpdate = true;

            // Update Head
            const dir = new THREE.Vector3().subVectors(currentPos, start);
            const length = dir.length();
            if (length > 0.1) {
                head.visible = true;
                head.position.copy(currentPos);

                const arrowDir = dir.clone().normalize();
                const axis = new THREE.Vector3(0, 1, 0).cross(arrowDir).normalize();
                const angle = Math.acos(new THREE.Vector3(0, 1, 0).dot(arrowDir));
                head.quaternion.setFromAxisAngle(axis, angle);

                const scale = (this.app.userWeight || 2) * 0.5;
                head.scale.set(scale, scale, scale);
            }
        }
        else if (this.app.activeTool === 'ANCHOR') {
            const dx = currentPos.x - start.x;
            const dz = currentPos.z - start.z;
            const angle = Math.atan2(dz, dx);
            if (this.currentShape) {
                this.currentShape.rotation.z = -angle;
            }
        }
        else if (this.app.activeTool === 'RECT') {
            const line = this.currentShape.getObjectByName('rect_visual');
            const plane = this.currentShape.getObjectByName('rect_hit_plane');

            // Update Line
            const points = [
                new THREE.Vector3(start.x, 0.5, start.z),
                new THREE.Vector3(currentPos.x, 0.5, start.z),
                new THREE.Vector3(currentPos.x, 0.5, currentPos.z),
                new THREE.Vector3(start.x, 0.5, currentPos.z)
            ];
            line.geometry.setFromPoints(points);

            // Update Hit Plane
            const centerX = (start.x + currentPos.x) / 2;
            const centerZ = (start.z + currentPos.z) / 2;
            const width = Math.abs(currentPos.x - start.x);
            const height = Math.abs(currentPos.z - start.z);

            plane.position.set(centerX, 0.5, centerZ);
            plane.scale.set(Math.max(0.1, width), Math.max(0.1, height), 1);
            plane.visible = true;
        }
        else if (this.app.activeTool === 'CIRCLE') {
            const radius = start.distanceTo(currentPos);
            const segments = 64;
            const points = [];
            for (let i = 0; i < segments; i++) {
                const theta = (i / segments) * Math.PI * 2;
                points.push(new THREE.Vector3(start.x + Math.cos(theta) * radius, 0.5, start.z + Math.sin(theta) * radius));
            }
            this.currentShape.geometry.setFromPoints(points);
        }
        else if (this.app.activeTool === 'PENCIL') {
            // Append new point
            const points = Array.from(this.currentShape.geometry.attributes.position.array);
            // BufferGeometry is static sized by default, need to create new one or use setFromPoints if we kept the points array
            // PENCIL is complex with BufferGeometry. Let's simplify: PENCIL creates a new Line segment for each move? 
            // Or easier: reconstruct geometry from tracked points.
            // We need to track points in instance.
            if (!this.currentShape.userData.points) this.currentShape.userData.points = [start];
            this.currentShape.userData.points.push(currentPos);
            this.currentShape.geometry.setFromPoints(this.currentShape.userData.points);
        }
    }

    finishDrawing(endPosition) {
        this.isDrawing = false;
        if (this.startPoint.distanceTo(endPosition) < 0.5 && this.app.activeTool !== 'PENCIL') {
            this.cancelDrawing();
            return;
        }

        const category = this.app.activeCategory;
        const key = this.app.selectedUnitKey;
        const isTelecom = category !== 'basic' && category !== undefined;
        // Include STRAND in this logic
        if (isTelecom && (['LINE', 'POINT', 'STRAND', 'TRENCH', 'BORE', 'ANCHOR', 'ARROW'].includes(this.app.activeTool))) {
            this.app.showSmartModal(category, key, (attributes) => {
                if (!attributes) {
                    this.cancelDrawing();
                    return;
                }

                const isWreckout = key === 'CABLE_WRECKOUT';
                if (isWreckout) {
                    this.addWreckoutMarkers(this.startPoint, endPosition);
                }

                // Resolve Unit to get Billing and Color
                const unitData = resolveSmartUnit(category, key, attributes);
                let finalColor = unitData.color;

                // Fallback Color Logic
                if (!finalColor) {
                    if (key === 'TRENCH') finalColor = '#FF0000';
                    else if (attributes.condition === 'New') finalColor = ASBUILT_THEME.NEW;
                    else if (attributes.condition === 'Remove') finalColor = ASBUILT_THEME.REMOVE;
                    else finalColor = ASBUILT_THEME.EXISTING;
                }

                // Force Update Mesh Color
                if (this.currentShape) {
                    this.currentShape.traverse(c => {
                        if (c.material) {
                            if (c.material.color) c.material.color.set(finalColor);
                            if (c.name === 'arrow_head') c.visible = true;
                        }
                    });
                    // Special case: Dashed Lines might need re-calc if something changed, but color set is enough
                    // If it's a LineDashedMaterial, standard .set works.
                }

                this.app.drawnObjects.push({
                    type: this.app.activeTool,
                    category: category,
                    baseKey: key,
                    attributes: attributes,
                    mesh: this.currentShape,
                    start: this.startPoint.clone(),
                    end: endPosition.clone(),
                    length: this.startPoint.distanceTo(endPosition),
                    color: finalColor
                });

                this.currentShape = null;
                this.app.updateTotals();
            });
        }
        else if (this.app.activeTool === 'DIMENSION') {
            // Calculate distance and add Text Label
            const dist = this.startPoint.distanceTo(endPosition).toFixed(1) + "'";

            // 1. The Line
            this.app.drawnObjects.push({
                type: 'DIMENSION_LINE',
                category: 'basic',
                start: this.startPoint.clone(),
                end: endPosition.clone(),
                mesh: this.currentShape,
                color: '#000000'
            });

            // 2. The Text (Centered)
            const mid = new THREE.Vector3().addVectors(this.startPoint, endPosition).multiplyScalar(0.5);
            this.placeText(mid, dist); // Use existing placeText for now, maybe style it smaller later

            this.currentShape = null;
        }
        else if (this.app.activeTool === 'CALLOUT') {
            // 1. Prompt for Text
            const text = prompt("Enter Note:", "");
            if (!text) {
                this.cancelDrawing();
                return;
            }

            // 2. The Leader Line
            this.app.drawnObjects.push({
                type: 'CALLOUT_LINE',
                category: 'basic',
                start: this.startPoint.clone(),
                end: endPosition.clone(),
                mesh: this.currentShape,
                color: '#000000'
            });

            // 3. The Text Box (at end point)
            // We'll use a specialized "BOXED_TEXT" style if possible, or just Text for now
            this.placeText(endPosition, text);

            this.currentShape = null;
        }
        else {
            // Basic tool: Persistent entry
            this.app.drawnObjects.push({
                type: this.app.activeTool,
                category: 'basic',
                baseKey: key,
                mesh: this.currentShape,
                start: this.startPoint.clone(),
                end: endPosition.clone(),
                color: '#000000', // STRICT ENFORCEMENT OF BLACK
                weight: this.app.userWeight || 2
            });
            this.currentShape = null;
            this.app.updateTotals();
        }
    }

    cancelDrawing() {
        this.isDrawing = false;
        if (this.currentShape) {
            this.app.scene.remove(this.currentShape);
            this.currentShape = null;
        }
        this.startPoint = null;
    }



    createSymbolSprite(shape, colorHex, label) {
        // Draw in WHITE so we can tint it with material color
        const drawColor = '#FFFFFF';

        const canvas = document.createElement('canvas');
        const ctx = canvas.getContext('2d');
        const size = 256; // Increased further for long text
        canvas.width = size; // Default width, might resize if label is long
        canvas.height = size; // Square default

        // If label is custom string (Removals), resize canvas to rectangle
        // SKIP resizing for POLE_X to keep it square!
        if (label && label.length > 3 && shape !== 'POLE_X') {
            canvas.width = 512;
            canvas.height = 128;
        } else if (shape === 'POLE_X') {
            // Force Square Large Canvas for Pole
            canvas.width = 512;
            canvas.height = 512;
        }

        const centerX = canvas.width / 2;
        const centerY = canvas.height / 2;
        const lineWidth = 2;

        ctx.strokeStyle = drawColor;
        ctx.fillStyle = drawColor;
        ctx.lineWidth = lineWidth;
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';

        // Specific Shapes First to avoid Label override
        // Specific Shapes First to avoid Label override
        if (shape === 'POLE_X') {
            // Circle with X
            ctx.lineWidth = 12; // Much Thicker lines for visibility
            ctx.beginPath();
            // Larger Circle (Radius 120) centered in 512x512
            // CenterY is 256. Shift up slightly for text space?
            // Let's keep it centered or slightly up.
            ctx.arc(centerX, centerY - 60, 120, 0, Math.PI * 2);
            ctx.stroke();

            // Large X
            // 45 degrees.
            const r = 85; // Length of arms from center
            ctx.beginPath();
            ctx.moveTo(centerX - r, (centerY - 60) - r);
            ctx.lineTo(centerX + r, (centerY - 60) + r);
            ctx.moveTo(centerX + r, (centerY - 60) - r);
            ctx.lineTo(centerX - r, (centerY - 60) + r);
            ctx.stroke();

            // Draw Tag Text Below
            if (label) {
                ctx.font = 'bold 80px Arial'; // Large Font
                ctx.fillStyle = colorHex; // Text in Color
                ctx.fillText(label, centerX, centerY + 160); // Position below circle
            }
        }
        // Custom Text Label Override (For Removals or non-shapes with long text)
        else if (label && label.length > 3) {
            ctx.font = 'bold 36px Arial'; // Slightly smaller for long labels
            ctx.fillText(label, centerX, centerY);
        }
        else if (shape === 'COPPER_C') {
            // "C" Symbol
            ctx.font = 'bold 40px Arial'; // Reduced size (User request: smaller font)
            ctx.fillText("C", centerX, centerY);
        }
        else if (shape === 'FIBER_F') {
            // "F" Symbol
            ctx.font = 'bold 40px Arial'; // Reduced size
            ctx.fillText("F", centerX, centerY);
        }
        else if (shape === 'SQUARE_DOT' || shape === 'PED') {
            ctx.strokeRect(centerX - s, centerY - s, s * 2, s * 2);
            ctx.font = 'bold 40px Arial';
            ctx.fillText("P", centerX, centerY);
        }
        else if (shape === 'HH_BOX') {
            ctx.font = 'bold 40px Arial';
            ctx.fillText("P", centerX, centerY);
        }
        else if (shape === 'rect_divided_H') {
            // Rectangle with center line and "H" on each side
            const w = size;
            const h = size / 2;
            const x = centerX - w;
            const y = centerY - h;

            // Box
            ctx.strokeRect(x, y, w * 2, h * 2);

            // Center Line
            ctx.beginPath();
            ctx.moveTo(centerX, y);
            ctx.lineTo(centerX, y + h * 2);
            ctx.stroke();

            // "H" on left and right
            ctx.font = 'bold 36px Arial';
            ctx.fillText("H", centerX - (w / 2), centerY + 12);
            ctx.fillText("H", centerX + (w / 2), centerY + 12);
        }
        else if (shape === 'square_x') {
            // Square with X
            const s = size / 1.5;
            const x = centerX - s;
            const y = centerY - s;

            // Box
            ctx.strokeRect(x, y, s * 2, s * 2);

            // X
            ctx.beginPath();
            ctx.moveTo(x, y);
            ctx.lineTo(x + s * 2, y + s * 2);
            ctx.moveTo(x + s * 2, y);
            ctx.lineTo(x, y + s * 2);
            ctx.stroke();
        }
        else if (shape === 'HH_BOX') {
            // Rectangle with HH
            const w = size / 2;
            const h = size / 3;
            ctx.strokeRect(centerX - w, centerY - h, w * 2, h * 2);
            ctx.font = 'bold 36px Arial';
            ctx.fillText("HH", centerX, centerY);
        }
        else if (shape === 'DIAMOND') {
            // Diamond for Splice
            ctx.beginPath();
            ctx.moveTo(centerX, centerY - 60);
            ctx.lineTo(centerX + 60, centerY);
            ctx.lineTo(centerX, centerY + 60);
            ctx.lineTo(centerX - 60, centerY);
            ctx.closePath();
            ctx.stroke();
            // Fill slightly?
            ctx.fillStyle = colorHex;
            ctx.globalAlpha = 0.2;
            ctx.fill();
            ctx.globalAlpha = 1.0;
        }
        else if (shape === 'RECT_LABEL') {
            // Vault/MPOP with Label
            const w = size / 2; // Wide box
            const h = size / 2;
            ctx.strokeRect(centerX - w, centerY - h, w * 2, h * 2);

            // Draw Tag inside if available (passed as label?)
            // If label is "V" or "MPOP", showing that is fine.
            // But if attributes.tag exists, we want THAT.
            // createSymbolSprite is called with (shape, color, unit_code OR text).
            // We need to pass the tag in the label.
            if (label) {
                ctx.font = 'bold 32px Arial';
                ctx.fillText(label, centerX, centerY);
            }
        }
        else if (shape === 'MPOP') {
            // Specific MPOP box? Or just RECT_LABEL
            const s = 50;
            ctx.strokeRect(centerX - s, centerY - s, s * 2, s * 2);
            ctx.font = 'bold 32px Arial';
            ctx.fillText("MPOP", centerX, centerY);
        }
        else if (shape === 'SQUARE') {
            const s = 40;
            ctx.fillRect(centerX - s, centerY - s, s * 2, s * 2);
        }
        else if (shape === 'ANCHOR_ARROW') {
            // Anchor Arrow >--|
            ctx.beginPath();
            ctx.moveTo(centerX - 40, centerY);
            ctx.lineTo(centerX + 40, centerY);
            // Head
            ctx.moveTo(centerX + 20, centerY - 20);
            ctx.lineTo(centerX + 40, centerY);
            ctx.lineTo(centerX + 20, centerY + 20);
            // Bar
            ctx.moveTo(centerX - 40, centerY - 20);
            ctx.lineTo(centerX - 40, centerY + 20);
            ctx.stroke();
        }
        else if (shape === 'SIMPLE_ARROW') {
            // Arrow ->
            ctx.beginPath();
            ctx.moveTo(centerX - 40, centerY);
            ctx.lineTo(centerX + 40, centerY);
            ctx.moveTo(centerX + 20, centerY - 20);
            ctx.lineTo(centerX + 40, centerY);
            ctx.lineTo(centerX + 20, centerY + 20);
            ctx.stroke();
        }
        else {
            // Default Circle
            ctx.beginPath();
            ctx.arc(centerX, centerY, 20, 0, Math.PI * 2);
            ctx.fill();
        }

        const texture = new THREE.CanvasTexture(canvas);
        const material = new THREE.SpriteMaterial({ map: texture, color: colorHex });
        const sprite = new THREE.Sprite(material);

        if (label && label.length > 3) {
            sprite.scale.set(30, 7.5, 1); // Wide rectangle for text
        } else {
            sprite.scale.set(10, 10, 1); // Standard square
        }

        return sprite;
    }





    recreateObject(data) {
        if (data.type === 'POINT') {
            const attributes = data.attributes || {};
            const unitData = resolveUnit(data.baseKey, attributes);
            let color = data.color || '#000000';
            if (attributes.condition === 'Remove') color = '#8B0000';
            else if (attributes.condition === 'New') color = '#008000';
            const shape = data.shape || 'SQUARE';

            const sprite = this.createSymbolSprite(shape, color, unitData.unit_code);
            const pos = new THREE.Vector3(data.position.x, data.position.y, data.position.z);
            sprite.position.copy(pos);
            sprite.position.y = 0.5;
            this.app.scene.add(sprite);

            // Restore Tag if present
            let tagSprite = null;
            if (attributes.tag) {
                tagSprite = this.createTextSprite(attributes.tag);
                tagSprite.position.copy(pos);
                tagSprite.position.z -= 5;
                tagSprite.position.y = 5;
                this.app.scene.add(tagSprite);
            }

            this.app.drawnObjects.push({
                type: 'POINT',
                category: data.category || 'aerial',
                baseKey: data.baseKey,
                attributes: attributes,
                unit_code: unitData.unit_code,
                position: pos,
                mesh: sprite,
                tagMesh: tagSprite,
                depth: data.depth
            });
        }
        else if (data.type === 'LINE') {

    addWreckoutMarkers(start, end) {
        const dist = start.distanceTo(end);
        const spacing = 20; // Every 20 feet
        const steps = Math.floor(dist / spacing);

        // Direction vector
        const dir = new THREE.Vector3().subVectors(end, start).normalize();

        for (let i = 1; i <= steps; i++) {
            const pos = start.clone().add(dir.clone().multiplyScalar(i * spacing));
            // Use Text 'X' for wreckout marker
            const textSprite = this.createTextSprite('X');
            textSprite.position.copy(pos);
            textSprite.position.y = 1;
            textSprite.scale.set(5, 5, 1); // Smaller than standard text

            // Override material color? createTextSprite uses Black text by default which matches Wreckout
            this.app.scene.add(textSprite);

            this.app.drawnObjects.push({
                type: 'MARKER',
                mesh: textSprite
            });
        }
    }

    createDashedTexture() {
        // Create 64x64 texture for repeating dash
        const canvas = document.createElement('canvas');
        canvas.width = 64;
        canvas.height = 64;
        const ctx = canvas.getContext('2d');

        // Transparent BG
        ctx.fillStyle = 'rgba(0,0,0,0)';
        ctx.fillRect(0, 0, 64, 64);

        // White Dash (left half) - Tintable
        ctx.fillStyle = '#FFFFFF';
        // Draw a rect that covers 50% of the width
        ctx.fillRect(0, 0, 32, 64);

        const THREE = window.THREE || this.app.THREE;
        const texture = new THREE.CanvasTexture(canvas);
        texture.wrapS = THREE.RepeatWrapping;
        texture.wrapT = THREE.RepeatWrapping;
        texture.minFilter = THREE.LinearFilter;
        return texture;
    }

    placeText(position, text) {
        const sprite = this.createTextSprite(text);
        sprite.position.copy(position);
        sprite.position.y = 1; // Above everything
        this.app.scene.add(sprite);

        this.app.drawnObjects.push({
            type: 'TEXT',
            text: text,
            position: position,
            mesh: sprite
        });
    }

    createTextSprite(message) {
        const fontface = "Arial";
        const fontsize = 128; // High resolution for sharp text
        // No border padding needed

        const canvas = document.createElement('canvas');
        const context = canvas.getContext('2d');

        // Dynamic size
        context.font = "Bold " + fontsize + "px " + fontface;
        const metrics = context.measureText(message);
        const textWidth = metrics.width;

        // Size canvas to fit text
        canvas.width = textWidth + 20;
        canvas.height = fontsize * 1.2;

        // Transparent Background (Default)
        // No Border

        // Text
        context.font = "Bold " + fontsize + "px " + fontface;
        context.fillStyle = "rgba(0, 0, 0, 1.0)";
        context.textAlign = "center";
        context.textBaseline = "middle";
        context.fillText(message, canvas.width / 2, canvas.height / 2);

        const texture = new THREE.CanvasTexture(canvas);
        texture.minFilter = THREE.LinearFilter;

        const material = new THREE.SpriteMaterial({ map: texture });
        const sprite = new THREE.Sprite(material);

        // Scale: Convert pixel size to World Units
        // 128px font -> we want this to be BIG in the world. 
        // Let's say we want the height to be roughly 15 units (bigger than pole)
        const aspect = canvas.width / canvas.height;
        const height = 5; // Target height in world units (Reduced)
        const width = height * aspect;

        sprite.scale.set(width, height, 1);

        return sprite;
    }
}

## BILLING RULES (inferred from descriptions)

1. **Most labor units are PER FOOT** (cable, strand, trench, bore, conduit placement)
2. **Point items are PER EACH** (poles, handholes, splice cases, pedestals, anchors)
3. **Splicing is PER COUNT** (number of pairs or fiber count determines which tier)
4. **Many jobs have BOTH a labor unit AND a material unit** — e.g.:
   - Pole job = `POLE WOOD <= 40ft` (labor EA) + `POLE 35-5 DF` (material EA)
   - Strand job = `STRAND 10M` (labor FT)

5. **Removal jobs are labor only** (no material)
6. **"EXISTING" items = $0 billing**, reference only
7. **Trench depth and conduit size determine the correct trench code** — not surface type
8. **Splice setup codes are charged separately** from the splice labor itself

- **Render**: Solid black line. "F" markers every ~200px. Count label.
  - NEW=Red, EXISTING=Black, REMOVE=Green
- **Modal fields**: Condition (NEW/EXISTING/REMOVE) | Count (12/24/48/72/96/144/216/288) | Footage | Start Foot Mark | End Foot Mark | Method (LASH/OVERLASH/ADSS/UG) | Rack/Slack footage
- **Billing output** (NEW):
  - Material: `FIB [count] 1JKT 1ARMOR LT` per FT (e.g. `FIB 144 1JKT 1ARMOR LT`)
  - Labor: `LASH FIBER` or `OVERLASH FIBER` per FT
  - If UG: `FIBER PL IN CONDUIT` per FT
  - If slack storage: `AERIAL FIBER SNOWSHOE` per EA
- **Billing output** (REMOVE):
  - `RMV AERIAL FIBER` per FT
- **Unit type**: FT (material and labor both per foot)

### COAX CABLE
- **Render**: Brown solid line. Size label.
- **Modal fields**: Size (.500/.625/.750/.875) | Footage | Method
- **Billing output**: No specific coax code in contract — flag for special quote
- **Unit type**: FT

## UNDERGROUND CATEGORY

### TRENCH
- **Render**: Red solid thick line. Label shows depth/width. "TRENCH" text.
- **Modal fields**: Cover Depth (30in/36in/48in) | Conduit Size (<=2.5in / 3-6in / >6in) | Type (Standard/Stick/Metal) | Footage
- **Billing output**: Correct code = `TRENCH [depth]in CVR [conduit size]` per FT
  - e.g. `TRENCH 30in CVR <= 2.5in`, `TRENCH 36in CVR 3-6in`, `TRENCH 48in CVR STICK <= 2.5in`
  - Open cut (no conduit): `OPEN TRENCH <= 2.5in` or `OPEN TRENCH 3-6in`
- **Unit type**: FT

### BORE
- **Render**: Brown dashed thick line. Label: "BORE [diameter]""
- **Modal fields**: Diameter (1.25/2/4 in) | Footage
- **Billing output**: No standard bore unit — maps to `SPECIAL QUOTE - BORE - LE` with supervisor flag
  - OR use open trench equivalent if applicable
- **Unit type**: FT (flagged)

### HANDHOLE (HH)
- **Render**: Rectangle divided by center vertical line, "H" on each side. Color by status.
- **Modal fields**: Status (NEW/EXISTING/REMOVE) | Dimensions (select from real sizes) | Lid Type (Standard/Split Lid) | Grounding (Y/N)
- **HH Size options** (real codes):
  - Small: `HH 13x24x18`, `HH 17x30x24`
  - Medium: `HH 30x48x36`, `HH 30x48x36 SPLIT LID`, `HH 30x60x30`
  - Large: `HH 36x60x36`, `HH 48x48x48`, `HH 48x60x36`
  - XL: `HH 48x72x48`, `HH 48x96x50`
- **Billing output**:
  - NEW: `HH [dimensions]` per EA (e.g. `HH 30x48x36`)
  - NEW+Grounding: also add `HH 13x24x18 GROUNDING` or `GROUND POLE ASSEMBLY`
  - REMOVE: `RMV HH` per EA
  - Grade adjust: `HH GRADE ADJUSTMENT` per EA
- **Unit type**: EA

### VAULT / MANHOLE (MH)
- **Render**: Larger rectangle with "V" or "MH" label.
- **Modal fields**: Status | Type (HH/MH) | Dimensions
- **MH codes**: `MH 48x72x84in`, `MH 72x72x72in`, `MH 72x144x84in`
- **Billing output**: MH size code per EA
- **Unit type**: EA

### SPLICE PIT
- **Render**: Square with X inside. Color by status.
- **Modal fields**: Status (NEW/REMOVE) | Size (SMALL/MEDIUM/LARGE)
- **Billing output**: `PIT SMALL 6x4x5ft`, `PIT MEDIUM 12x6x5ft`, or `PIT LARGE 12x12x6ft` per EA
- **Unit type**: EA

### PEDESTAL
- **Render**: Small rectangle with dot inside. "PED" label.
- **Modal fields**: Status | Type (Copper/Fiber) | Size (6/8/10/12/14in) | Mount (Stake/Pole/Integral Stake)
- **Billing output**:
  - Copper: `PED [size]in [mount type]` e.g. `PED 10in STAKE MOUNT`
  - Fiber: `PED FIB [size]in [mount type]` e.g. `PED FIB 10in STAKE MOUNT`
  - Sizes map: SMALL=6-12in, MEDIUM=14-24in, LARGE=36-42in
  - REMOVE: `RMV PEDESTAL` per EA
- **Unit type**: EA

### CONDUIT
- **Render**: Dashed gray line. Size label. "CDUT" text.
- **Modal fields**: Size (1in/2in/3in/4in) | Schedule (SCH40/SCH80/HDPE) | Footage | Location (Aerial/UG)
- **Billing output**:
  - Material: `CONDUIT 2in SCH40 20ft GRY` or `CONDUIT 2in SCH40 20ft GRY UG` per EA (per stick)
  - Labor placement: `HDPE PL IN CONDUIT <= 2in` or `FIBER IN CONDUIT ISP` per FT
- **Unit type**: EA (material per stick) + FT (labor)

## SPLICING CATEGORY

### SPLICE CASE
- **Render**: Diamond shape. Color by status. Case ID label.
- **Modal fields**: Status (NEW/EXISTING) | Location (AERIAL/BURIED/UG) | Cable Type (COPPER/FIBER) | Case Size (from dropdown of real codes) | Pressure (Pressurized/Non-pressure/Free Breathing)
- **Billing output**:
  - NEW FIBER: `CASE FIBER NEW` (labor) + specific case material code per EA
  - NEW COPPER: `CASE COPPER NEW PRESSURE` or `CASE COPPER NEW NON-PRESSURE` (labor) + case material
  - RE-ENTER: `RE-ENTER EXIST FIBER CASE` or `RE-ENTER EXIST COPPER CASE`
  - Always add: `SPL SETUP-TEARDOWN [location]-[type]` per EA
    - e.g. `SPL SETUP-TEARDOWN AERIAL-FIB`, `SPL SETUP-TEARDOWN HH-COP`
- **Unit type**: EA

### SPLICE ACTIVITY (Fiber)
- **Render**: Small circle/dot on splice case. Count label.
- **Modal fields**: Type (FUSION/RIBBON/MECHANICAL) | Count (number of fibers/pairs)
- **Billing output** (Fiber Fusion):
  - 1-4 fibers: `SPLICE FIBER FUSION 1-4`
  - 5-12: `SPLICE FIBER FUSION 5-12`
  - 13-24: `SPLICE FIBER FUSION 13-24`
  - 25-48: `SPLICE FIBER FUSION 25-48`
  - 49-144: `SPLICE FIBER FUSION 49-144`
  - 145-288: `SPLICE FIBER FUSION 145-288`
  - 289-432: `SPLICE FIBER FUSION 289-432`
  - 433-864: `SPLICE FIBER FUSION 433-864`
- **Billing output** (Ribbon):
  - <=2 ribbons: `SPLICE FIBER RIBBONS <=2`
  - 3-12: `SPLICE FIBER RIBBONS 3-12`, etc.
- **Unit type**: EA (per splice count tier)

### SPLICE ACTIVITY (Copper)
- **Billing output**:
  - Mechanical 1-25pr: `SPLICE COPPER MECHANICAL 1-25pr`
  - Mechanical 26-100pr: `SPLICE COPPER MECHANICAL 26-100pr`
  - Mechanical >100pr: `SPLICE COPPER MECHANICAL > 100pr`
  - Module <=300pr: `SPLICE COPPER MODULE <= 300pr`
  - Module 301-900pr: `SPLICE COPPER MODULE 301-900pr`
  - Module 901-1800pr: `SPLICE COPPER MODULE 901-1800pr`
  - Module >1800pr: `SPLICE COPPER MODULE > 1800pr`
- **Unit type**: EA

### TERMINAL
- **Render**: Small square. Label with pair count.
- **Modal fields**: Type | Pair count
- **Billing output**: Terminal-specific codes (part of case install)

### MPOP
- **Render**: Rectangle with "MPOP" label.
- **Modal fields**: Label text
- **Billing output**: Reference only / Special Quote

### RISER
- **Render**: Vertical rectangle at pole base. Size label.
- **Modal fields**: Diameter (2in/3in/4in) | Material (Plastic/Steel) | Height
- **Billing output**:
  - Material: `RISER PIPE POLE 2in PLASTIC` or `2in STEEL` or `3in PLASTIC` etc.
  - Guard: `RISER U-GUARD 2 in PLASTIC` or `3in PLASTIC` etc.
  - Removal: `RMV POLE RISER OR U-GUARD`
- **Unit type**: EA

### GROUND ROD
- **Render**: Small circle with downward bar.
- **Modal fields**: Size (5/8in-8ft / 5/8in-20ft / 1/2in-5ft)
- **Billing output**:
  - `GROUND ROD 5/8in-8ft` (most common)
  - `GROUND ROD 5/8in-20ft THREADED`
  - `GROUND ROD 1/2in-5ft`
  - Also: `GROUND POLE ASSEMBLY` (for HH grounding)
- **Unit type**: EA
